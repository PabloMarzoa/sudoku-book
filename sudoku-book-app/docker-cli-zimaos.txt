# OPCIÓN 1: Usando Docker CLI desde la terminal (ZimaOS SSH)

# 1. Sitúate en la carpeta donde has copiado el proyecto y construye la imagen:
docker build -t sudoku-book-app .

# 2. Arranca el contenedor:
docker run -d \
  --name sudoku-book-app \
  -p 8080:80 \
  --restart unless-stopped \
  sudoku-book-app
